Overview — How the system works

The dashboard is a React SPA living inside bpl-tools/Admin/. It exports ready-made page components. Your plugin consumes them by:

  1. Creating a data.js file with your plugin's metadata
  2. Building a lightweight React App.js that routes between pages
  3. Registering a WordPress admin page and enqueuing the compiled bundle
Single source of truth: All plugin-specific content (name, slugs, changelogs, getting-started steps, demos, pricing) lives in your data.js file. The shared components in bpl-tools are never modified per-plugin.

Create your plugin's data file

Create src/admin/utils/data.js. This is the single source of truth for all dashboard content.

Changelog badge color rules

The badge is injected automatically from the <strong> text at the start of each list item. You only need to prefix items correctly.

Leading <strong> text Badge color Example
New, Add New Primary / Blue '<strong>New:</strong> Feature description.'
Update Update Orange #FF8D28 '<strong>Update:</strong> SDK updated.'
Fix, Fixing, Fixed, Remove Fix Red #FF383C '<strong>Fix:</strong> Resolved the issue.'
Improve, Improvement Improvement Purple #8b5cf6 '<strong>Improvement:</strong> Better UX.'
Anything else Other Gray #9ca3af Rarely needed.
Important: Colons inside the badge are stripped automatically (Fix: renders as "Fix"). Always include the colon — it's conventional and then removed cleanly.

Create your blocks data file

Create src/admin/utils/blocks.js:

import { myBlockIcon, anotherIcon } from './blocksIcons';

const pluginSlug = 'your-plugin';
const demoURL		= 'https://yourdomain.com/demo';
const docsURL		= 'https://bplugins.com/docs/your-plugin';

export default [
	// ── Grouped block (renders with a section heading) ─────────────────
	{
		title: 'Layout',
		children: [
			{
				name:		 `${pluginSlug}/container`,
				title:		'Container',
				icon:		 myBlockIcon,
				demo:		 `${demoURL}/container/`,
				docs:		 `${docsURL}/container-block/`,
				status:	 'published',
				required: true		 // cannot be disabled; shows "Required" badge
			}
		]
	},

	// ── Individual block ───────────────────────────────────────────────
	{
		name:		`${pluginSlug}/my-block`,
		title:	 'My Block',
		icon:		anotherIcon,
		demo:		`${demoURL}/my-block/`,
		docs:		`${docsURL}/my-block/`,
		status:	'published',
		badge:	 'New'				 // optional text badge on the card
	},

	// ── Pro-only block ─────────────────────────────────────────────────
	{
		name:			`${pluginSlug}/pro-block`,
		title:		 'Pro Block',
		icon:			anotherIcon,
		status:		'published',
		isPremium: true		 // locks toggle, shows "Pro" badge, opens upgrade modal
	}
];
Field Type Notes
name string Gutenberg block name, e.g. plugin/block-name
title string Display name on the card
icon ReactNode SVG rendered in the colored circle
demo string URL → demo link icon on the card
docs string URL → docs link icon on the card
status string 'published' or omit (drafts are hidden)
badge string Optional custom text badge (top-left)
required boolean Locks the toggle; shows "Required" badge
isPremium boolean Locks toggle; shows "Pro" badge; opens ProModal
children array Makes it a group with a section heading above

Build the App router

Create src/admin/Components/App.js:

import { useState } from 'react';
import { HashRouter as Router, Routes, Route, Navigate } from 'react-router-dom';

import Welcome			from '../../../../bpl-tools/Admin/Welcome';
import Demos				from '../../../../bpl-tools/Admin/Demos';
import Pricing			from '../../../../bpl-tools/Admin/Pricing';
import Activation	 from '../../../../bpl-tools/Admin/Activation';
import OurPlugins	 from '../../../../bpl-tools/Admin/OurPlugins';

import Layout from './Layout';	 // wraps Header + nav
import { demoInfo, pricingInfo, welcomeInfo } from '../utils/data';

const App = (props) => {
	const { isPremium, adminUrl } = props;

	return <Router>
		<Routes>
			<Route path='/' element={<Layout {...props} />}>

				<Route index element={<Welcome {...props} {...welcomeInfo(adminUrl)} />} />
				<Route path='welcome'
							 element={<Welcome {...props} {...welcomeInfo(adminUrl)} />} />

				<Route path='demos'
							 element={<Demos {...props} demoInfo={demoInfo} />} />

				{!isPremium && <Route path='pricing'
							 element={<Pricing pricingInfo={pricingInfo} options={{}} {...props} />} />}

				<Route path='activation' element={<Activation {...props} />} />
				<Route path='our-plugins' element={<OurPlugins {...props} />} />

				<Route path='*' element={<Navigate to='/welcome' replace />} />
			</Route>
		</Routes>
	</Router>;
};
export default App;
Welcome spread pattern: Always spread props (from dashboardInfo) first, then welcomeInfo(adminUrl). The second spread adds gettingStarted, changelogs, proFeatures etc.

Admin entry point

Create src/admin/index.js:

import { render } from '@wordpress/element';
import App from './Components/App';

import '../../../../bpl-tools/Admin/style.scss';	// base dashboard styles
import './style.scss';														// your CSS variable overrides

const root = document.getElementById('your-plugin-admin-root');
if (root) render(<App info={window.yourPluginAdmin} />, root);
Import bpl-tools/Admin/style.scss first, then your own style.scss. This order ensures your CSS variable declarations override the fallback values in the shared stylesheet.

Set CSS custom properties (theming)

Every component uses CSS variables for colors. Setting them once in your root SCSS instantly themes the entire dashboard — icon tints, badges, buttons, borders, and scrollbars.

Create / update src/admin/style.scss:

// Replace values with your plugin's brand colors
.bPlDashboard {
	--bpl-dashboard-primary-color:			 #146EF5;
	--bpl-dashboard-primary-color-rgb:	 20, 110, 245;	// same, as R G B
	--bpl-dashboard-secondary-color:		 #FF7A00;
	--bpl-dashboard-title-color:				 #070127;
	--bpl-dashboard-content-color:			 #485781;
	--bpl-dashboard-content-color-rgb:	 72, 87, 129;
}
The -rgb variants are required. They are consumed in expressions like rgba(var(--bpl-dashboard-primary-color-rgb, ...), 0.1) to produce transparent tints. CSS does not support extracting channels from a hex value at runtime.

Where each variable is used

Variable Used for
--bpl-dashboard-primary-color Block badge, "New" changelog badge, buttons, block border accents
--bpl-dashboard-primary-color-rgb Icon background tint (10% opacity), block card hover border
--bpl-dashboard-secondary-color Button hover fill, changelog left-border for "update" type
--bpl-dashboard-title-color All heading text
--bpl-dashboard-content-color Body / paragraph text
--bpl-dashboard-content-color-rgb Scrollbar thumb, dimmed time text (75% opacity)

add_action('admin_menu', function () {

Enqueue scripts + localize data

add_action('admin_enqueue_scripts', function ($hook) {
	if ($hook !== 'toplevel_page_your-plugin') return;

	wp_enqueue_script(
		'your-plugin-admin',
		plugin_dir_url(__FILE__) . 'build/admin.js',
		['wp-element', 'wp-i18n', 'wp-api-fetch'],
		YOUR_PLUGIN_VERSION,
		true
	);

	wp_localize_script('your-plugin-admin', 'yourPluginAdmin', [
		'version'						=> YOUR_PLUGIN_VERSION,
		'isPremium'					=> your_is_premium_check(),
		'nonce'							=> wp_create_nonce('wp_ajax'),
		'adminUrl'					 => admin_url(),
		'disabledBlocks'		 => get_option('bPlBlocksDisabled', []),
		'licenseActiveNonce' => wp_create_nonce('your_license_nonce'),
	]);
});

AJAX handler — save disabled blocks

add_action('wp_ajax_bPlBlocksDisabled', function () {
	$nonce = sanitize_text_field(wp_unslash($_POST['_wpnonce'] ?? ''));
	if (!wp_verify_nonce($nonce, 'wp_ajax')) {
		wp_send_json_error('Invalid Request');
	}
	$data		= json_decode(stripslashes($_POST['data']), true);
	$db_data = get_option('bPlBlocksDisabled', []);

	if (!isset($data) && $db_data) {
		wp_send_json_success($db_data);
	}
	update_option('bPlBlocksDisabled', $data);
	wp_send_json_success($data);
});

Final checklist

  • data.js — dashboardInfo filled with real plugin name, slugs, media URLs, freemius IDs
